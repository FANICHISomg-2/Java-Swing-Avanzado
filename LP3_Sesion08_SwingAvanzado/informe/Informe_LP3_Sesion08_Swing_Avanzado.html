EJERCICIO 1: Gestión de producto (Binding de datos manual)

Enunciado: aplicación que ingresa nombre, precio, stock y categoría de un producto; al pulsar "Actualizar Producto" se actualiza el modelo y se muestra la información en un JLabel.

Clases usadas: Producto (modelo), JTextField, JButton, JLabel, ActionListener.

import javax.swing.*;
import java.awt.*;

class Producto {
    private String nombre;
    private double precio;
    private int cantidadStock;
    private String categoria;

    public Producto(String nombre, double precio, int cantidadStock, String categoria) {
        this.nombre = nombre; this.precio = precio;
        this.cantidadStock = cantidadStock; this.categoria = categoria;
    }
    public String getNombre() { return nombre; }
    public void setNombre(String nombre) { this.nombre = nombre; }
    public double getPrecio() { return precio; }
    public void setPrecio(double precio) { this.precio = precio; }
    public int getCantidadStock() { return cantidadStock; }
    public void setCantidadStock(int cantidadStock) { this.cantidadStock = cantidadStock; }
    public String getCategoria() { return categoria; }
    public void setCategoria(String categoria) { this.categoria = categoria; }
}

public class AppProducto {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Gestión de Producto");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(380, 330);

        Producto producto = new Producto("Laptop", 2500.0, 10, "Tecnología");

        JTextField txtNombre = new JTextField(producto.getNombre(), 15);
        JTextField txtPrecio = new JTextField(String.valueOf(producto.getPrecio()), 15);
        JTextField txtStock = new JTextField(String.valueOf(producto.getCantidadStock()), 15);
        JTextField txtCategoria = new JTextField(producto.getCategoria(), 15);
        JButton btn = new JButton("Actualizar Producto");
        JLabel lblInfo = new JLabel("Información del producto aparecerá aquí");

        btn.addActionListener(e -> {
            try {
                producto.setNombre(txtNombre.getText().trim());
                producto.setPrecio(Double.parseDouble(txtPrecio.getText().trim()));
                producto.setCantidadStock(Integer.parseInt(txtStock.getText().trim()));
                producto.setCategoria(txtCategoria.getText().trim());
                lblInfo.setText("<html><b>Producto actualizado</b><br>"
                    + "Nombre: " + producto.getNombre() + "<br>"
                    + "Precio: S/ " + producto.getPrecio() + "<br>"
                    + "Stock: " + producto.getCantidadStock() + "<br>"
                    + "Categoría: " + producto.getCategoria() + "</html>");
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(frame, "Precio o stock inválido. Ingrese números.");
            }
        });

        frame.setLayout(new FlowLayout());
        frame.add(new JLabel("Nombre:"));    frame.add(txtNombre);
        frame.add(new JLabel("Precio:"));    frame.add(txtPrecio);
        frame.add(new JLabel("Stock:"));     frame.add(txtStock);
        frame.add(new JLabel("Categoría:")); frame.add(txtCategoria);
        frame.add(btn);
        frame.add(lblInfo);
        frame.setVisible(true);
    }
}

Explicación: el botón lee los valores de los campos (getText), los convierte y los envía al modelo con los set; luego la etiqueta se rellena leyendo el modelo con los get. Se valida con try/catch de NumberFormatException.

EJERCICIO 2: Gráfico de líneas de temperaturas semanales

Enunciado: siete campos (lunes a domingo), botón "Mostrar Gráfico" y un panel que dibuja puntos unidos por una línea.

Clases usadas: JPanel, paintComponent, Graphics2D (drawLine, fillOval, drawString, antialiasing, setStroke).

import javax.swing.*;
import java.awt.*;

class PanelGrafico extends JPanel {
    private double[] temps = new double[7];
    private final String[] dias = {"Lun", "Mar", "Mié", "Jue", "Vie", "Sáb", "Dom"};

    public void setTemperaturas(double[] t) { this.temps = t; repaint(); }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2 = (Graphics2D) g;
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        int mI = 50, mD = 20, mS = 20, mInf = 40;
        int w = getWidth() - mI - mD, h = getHeight() - mS - mInf;

        double min = temps[0], max = temps[0];
        for (double t : temps) { min = Math.min(min, t); max = Math.max(max, t); }
        if (max == min) { max += 1; min -= 1; }

        // Ejes
        g2.setColor(Color.BLACK);
        g2.drawLine(mI, mS, mI, mS + h);
        g2.drawLine(mI, mS + h, mI + w, mS + h);
        g2.drawString(String.format("%.1f", max), 8, mS + 5);
        g2.drawString(String.format("%.1f", min), 8, mS + h);
        g2.drawString("°C", 8, mS - 5 + 10);

        // Puntos y línea
        int[] xs = new int[7], ys = new int[7];
        for (int i = 0; i < 7; i++) {
            xs[i] = mI + 20 + i * (w - 40) / 6;
            ys[i] = (int) (mS + h - (temps[i] - min) / (max - min) * h);
        }
        g2.setColor(new Color(30, 90, 200));
        g2.setStroke(new BasicStroke(2f));
        for (int i = 0; i < 6; i++) g2.drawLine(xs[i], ys[i], xs[i + 1], ys[i + 1]);

        for (int i = 0; i < 7; i++) {
            g2.setColor(Color.RED);
            g2.fillOval(xs[i] - 5, ys[i] - 5, 10, 10);
            g2.setColor(Color.BLACK);
            g2.drawString(String.valueOf(temps[i]), xs[i] - 10, ys[i] - 10);
            g2.drawString(dias[i], xs[i] - 10, mS + h + 20);
        }
    }
}

public class AppTemperaturas {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Temperaturas de la semana");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(650, 480);
        frame.setLayout(new BorderLayout());

        String[] dias = {"Lun", "Mar", "Mié", "Jue", "Vie", "Sáb", "Dom"};
        JTextField[] campos = new JTextField[7];
        JPanel top = new JPanel();
        for (int i = 0; i < 7; i++) {
            campos[i] = new JTextField("0", 3);
            top.add(new JLabel(dias[i] + ":"));
            top.add(campos[i]);
        }
        JButton btn = new JButton("Mostrar Gráfico");
        top.add(btn);

        PanelGrafico panel = new PanelGrafico();
        btn.addActionListener(e -> {
            try {
                double[] t = new double[7];
                for (int i = 0; i < 7; i++) t[i] = Double.parseDouble(campos[i].getText().trim());
                panel.setTemperaturas(t);
            } catch (NumberFormatException ex) {
                JOptionPane.showMessageDialog(frame, "Ingrese temperaturas numéricas válidas.");
            }
        });

        frame.add(top, BorderLayout.NORTH);
        frame.add(panel, BorderLayout.CENTER);
        frame.setVisible(true);
    }
}

Explicación: el panel escala las temperaturas entre el mínimo y el máximo para ubicar cada punto; setTemperaturas guarda los datos y llama a repaint(), que invoca paintComponent y actualiza el gráfico.

EJERCICIO 3: Reproductor de efectos de sonido

Enunciado: botones "Aplausos", "Campana" y "Explosión" que reproducen un WAV distinto con Clip.

Clases usadas: AudioSystem, AudioInputStream, Clip (javax.sound.sampled).

import javax.sound.sampled.*;
import javax.swing.*;
import java.awt.*;
import java.io.File;

public class AppEfectos {
    public static void main(String[] args) {
        JFrame frame = new JFrame("Efectos de Sonido");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(380, 120);
        frame.setLayout(new FlowLayout());

        JButton bAplausos = new JButton("Aplausos");
        JButton bCampana = new JButton("Campana");
        JButton bExplosion = new JButton("Explosión");

        bAplausos.addActionListener(e -> reproducir("aplausos.wav"));
        bCampana.addActionListener(e -> reproducir("campana.wav"));
        bExplosion.addActionListener(e -> reproducir("explosion.wav"));

        frame.add(bAplausos); frame.add(bCampana); frame.add(bExplosion);
        frame.setVisible(true);
    }

    public static void reproducir(String ruta) {
        try {
            AudioInputStream audio = AudioSystem.getAudioInputStream(new File(ruta));
            Clip clip = AudioSystem.getClip();
            clip.open(audio);
            clip.start();
        } catch (Exception ex) {
            JOptionPane.showMessageDialog(null, "No se pudo reproducir: " + ruta
                + "\n(" + ex.getMessage() + ")");
        }
    }
}
Coloca aplausos.wav, campana.wav y explosion.wav en la raíz del proyecto de Eclipse (junto a src), como se hizo con sample-3s.wav en la guía. Deben ser WAV PCM.

EJERCICIO 4: Reproducir, pausar y reanudar música

Enunciado: tres botones que controlan una pista de audio: reproducir desde el inicio, pausar y reanudar desde donde se detuvo.

Clases usadas: Clip con setMicrosecondPosition, getMicrosecondPosition, start, stop.

import javax.sound.sampled.*;
import javax.swing.*;
import java.awt.*;
import java.io.File;

public class AppReproductor {
    private static Clip clip;
    private static long posicion = 0;

    public static void main(String[] args) {
        JFrame frame = new JFrame("Reproductor de Música");
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setSize(350, 120);
        frame.setLayout(new FlowLayout());

        JButton bPlay = new JButton("Reproducir");
        JButton bPausa = new JButton("Pausar");
        JButton bReanudar = new JButton("Reanudar");

        bPlay.addActionListener(e -> reproducir("musica.wav"));
        bPausa.addActionListener(e -> pausar());
        bReanudar.addActionListener(e -> reanudar());

        frame.add(bPlay); frame.add(bPausa); frame.add(bReanudar);
        frame.setVisible(true);
    }

    static void reproducir(String ruta) {
        try {
            if (clip != null) { clip.stop(); clip.close(); }
            AudioInputStream audio = AudioSystem.getAudioInputStream(new File(ruta));
            clip = AudioSystem.getClip();
            clip.open(audio);
            clip.setMicrosecondPosition(0);   // desde el inicio
            clip.start();
            posicion = 0;
        } catch (Exception ex) {
            JOptionPane.showMessageDialog(null, "Error al cargar el audio: " + ex.getMessage());
        }
    }

    static void pausar() {
        if (clip != null && clip.isRunning()) {
            posicion = clip.getMicrosecondPosition();  // guarda dónde se detuvo
            clip.stop();
        }
    }

    static void reanudar() {
        if (clip != null && !clip.isRunning()) {
            clip.setMicrosecondPosition(posicion);     // continúa desde ahí
            clip.start();
        }
    }
}
Coloca musica.wav en la raíz del proyecto. La pausa funciona guardando la posición con getMicrosecondPosition() antes de stop(), y la reanudación la restaura con setMicrosecondPosition().

CONCLUSIONES

  • El binding manual se logra con clases modelo, listeners en los controles y el uso de get/set para sincronizar vista y modelo.
  • Graphics y Graphics2D sobre paintComponent permiten dibujar gráficos de datos que se actualizan con repaint().
  • Clip de javax.sound.sampled permite reproducir efectos y controlar la reproducción (pausa y reanudación) de audio WAV.
